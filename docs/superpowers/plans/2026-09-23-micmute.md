# MicMute Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A macOS menu bar app that mutes/unmutes every microphone (built-in, USB, Bluetooth), driven by a status item, global hotkeys (toggle + push-to-talk), and an Elgato Stream Deck plugin.

**Architecture:** SwiftPM package with a testable `MicMuteCore` library (CoreAudio wrapper, mute logic, preferences, WebSocket bridge) and a thin AppKit/SwiftUI `MicMute` executable assembled into a `.app` by a Makefile. A TypeScript Stream Deck plugin talks to the app over a loopback WebSocket (`127.0.0.1:52836`); the app is the single source of truth.

**Tech Stack:** Swift 6 toolchain (Swift 5 language mode), CoreAudio, Network.framework, Carbon hotkeys, AppKit, SwiftUI, ServiceManagement, Swift Testing. Plugin: TypeScript 5.9, `@elgato/streamdeck` v3, Rollup, Vitest, `@elgato/cli`.

**Spec:** `docs/superpowers/specs/2026-09-23-micmute-design.md`

```swift
/// Single source of truth for whether the microphones should be muted.
/// Every trigger (status item, hotkeys, Stream Deck) goes through here.
@MainActor
public final class MuteState {
    public struct Snapshot: Equatable, Sendable {
        public let muted: Bool
        public let pttActive: Bool

        public init(muted: Bool, pttActive: Bool) {
            self.muted = muted
            self.pttActive = pttActive
        }
    }

    public struct Change: Equatable, Sendable {
        public let snapshot: Snapshot
        public let mutedChanged: Bool
        public let viaPTT: Bool

        public init(snapshot: Snapshot, mutedChanged: Bool, viaPTT: Bool) {
            self.snapshot = snapshot
            self.mutedChanged = mutedChanged
            self.viaPTT = viaPTT
        }
    }

    /// The user's toggle state, ignoring push-to-talk.
    public private(set) var isMuted: Bool
    private var pttSources: Set<AnyHashable> = []
    private let apply: (Bool) -> Void
    private var observers: [(Change) -> Void] = []

    public init(initiallyMuted: Bool, apply: @escaping (Bool) -> Void) {
        isMuted = initiallyMuted
        self.apply = apply
    }

    public var pttActive: Bool { !pttSources.isEmpty }
    public var effectiveMuted: Bool { isMuted && !pttActive }
    public var snapshot: Snapshot { Snapshot(muted: effectiveMuted, pttActive: pttActive) }

    public func addObserver(_ observer: @escaping (Change) -> Void) {
        observers.append(observer)
    }

    public func toggle() { setMuted(!isMuted) }

    public func setMuted(_ muted: Bool) {
        update(viaPTT: false) { isMuted = muted }
    }

    public func pttBegin(source: AnyHashable) {
        update(viaPTT: true) { pttSources.insert(source) }
    }

    public func pttEnd(source: AnyHashable) {
        update(viaPTT: true) { pttSources.remove(source) }
    }

    private func update(viaPTT: Bool, _ mutate: () -> Void) {
        let before = snapshot
        mutate()
        let after = snapshot
        guard after != before else { return }
        let mutedChanged = after.muted != before.muted
        if mutedChanged { apply(after.muted) }
        let change = Change(snapshot: after, mutedChanged: mutedChanged, viaPTT: viaPTT)
        observers.forEach { $0(change) }
    }
}
```

```swift
/// Abstraction over the CoreAudio HAL so mute logic can be tested with a fake.
@MainActor
public protocol AudioSystem: AnyObject {
    /// Called (on the main queue) when the device list or any input device's mute/volume changes.
    var onChange: (() -> Void)? { get set }

    func inputDeviceIDs() -> [DeviceID]
    func name(of id: DeviceID) -> String
    func uid(of id: DeviceID) -> String

    func canSetMute(_ id: DeviceID) -> Bool
    func isMuted(_ id: DeviceID) -> Bool
    @discardableResult func setMuted(_ id: DeviceID, _ muted: Bool) -> Bool

    func canSetVolume(_ id: DeviceID) -> Bool
    func volume(_ id: DeviceID) -> Float
    @discardableResult func setVolume(_ id: DeviceID, _ volume: Float) -> Bool
}
```

```swift
public enum MuteMethod: Equatable, Sendable {
    case mute, volume, unsupported
}

public struct DeviceStatus: Equatable, Sendable {
    public let name: String
    public let method: MuteMethod
    public let isMuted: Bool

    public init(name: String, method: MuteMethod, isMuted: Bool) {
        self.name = name
        self.method = method
        self.isMuted = isMuted
    }

    public var summary: String {
        let state = isMuted ? "Muted" : "Live"
        switch method {
        case .mute: return "\(name) — \(state)"
        case .volume: return "\(name) — \(state) (via volume)"
        case .unsupported: return "\(name) — ⚠︎ can't mute"
        }
    }
}

/// Applies the desired mute state to every input device and keeps it applied while muted.
/// Devices without a mute control are muted by setting input volume to 0 and restoring it later.
@MainActor
public final class MicController {
    private let audio: AudioSystem
    private var savedVolumes: [String: Float] = [:]
    public private(set) var desiredMuted = false

    public init(audio: AudioSystem) {
        self.audio = audio
        audio.onChange = { [weak self] in self?.enforce() }
    }

    public func apply(muted: Bool) {
        desiredMuted = muted
        audio.inputDeviceIDs().forEach { apply(muted, to: $0) }
    }

    public var deviceStatuses: [DeviceStatus] {
        audio.inputDeviceIDs().map { id in
            let name = audio.name(of: id)
            if audio.canSetMute(id) { return DeviceStatus(name: name, method: .mute, isMuted: audio.isMuted(id)) }
            if audio.canSetVolume(id) { return DeviceStatus(name: name, method: .volume, isMuted: audio.volume(id) == 0) }
            return DeviceStatus(name: name, method: .unsupported, isMuted: false)
        }
    }

    /// True when at least one device can be muted and every such device currently is.
    public var allMuted: Bool {
        let controllable = deviceStatuses.filter { $0.method != .unsupported }
        return !controllable.isEmpty && controllable.allSatisfy(\.isMuted)
    }

    private func enforce() {
        guard desiredMuted else { return }
        audio.inputDeviceIDs().forEach { apply(true, to: $0) }
    }

    private func apply(_ muted: Bool, to id: DeviceID) {
        if audio.canSetMute(id) {
            if audio.isMuted(id) != muted { audio.setMuted(id, muted) }
        } else if audio.canSetVolume(id) {
            muted ? muteByVolume(id) : unmuteByVolume(id)
        }
    }

    private func muteByVolume(_ id: DeviceID) {
        let current = audio.volume(id)
        guard current > 0 else { return }
        savedVolumes[audio.uid(of: id)] = current
        audio.setVolume(id, 0)
    }

    private func unmuteByVolume(_ id: DeviceID) {
        if let saved = savedVolumes.removeValue(forKey: audio.uid(of: id)) {
            audio.setVolume(id, saved)
        } else if audio.volume(id) == 0 {
            audio.setVolume(id, 1)
        }
    }
}
```

```swift
import CoreAudio
import Foundation
import os

/// CoreAudio HAL implementation of `AudioSystem`. Listener callbacks are delivered on the main queue.
///
/// Mute/volume use the input scope's main element when it is settable, otherwise each input channel
/// element (1...n) that is settable — some USB interfaces only expose per-channel controls.
@MainActor
public final class CoreAudioSystem: AudioSystem {
    public var onChange: (() -> Void)?

    private let log = Logger(subsystem: "xyz.kayvan.micmute", category: "audio")
    private var watchedDevices: Set<DeviceID> = []
    private lazy var listener: AudioObjectPropertyListenerBlock = { [weak self] _, _ in
        MainActor.assumeIsolated { self?.handleChange() }
    }

    public init() {
        var address = Self.address(kAudioHardwarePropertyDevices, scope: kAudioObjectPropertyScopeGlobal)
        let status = AudioObjectAddPropertyListenerBlock(AudioObjectID(kAudioObjectSystemObject), &address, .main, listener)
        if status != noErr { log.error("Device list listener failed: \(status)") }
        watchInputDevices()
    }

    public func inputDeviceIDs() -> [DeviceID] {
        allDeviceIDs().filter { inputChannelCount($0) > 0 }
    }

    public func name(of id: DeviceID) -> String {
        string(id, kAudioObjectPropertyName) ?? "Unknown device"
    }

    public func uid(of id: DeviceID) -> String {
        string(id, kAudioDevicePropertyDeviceUID) ?? "device-\(id)"

    public func canSetMute(_ id: DeviceID) -> Bool {
        !settableElements(id, kAudioDevicePropertyMute).isEmpty
    }

    public func isMuted(_ id: DeviceID) -> Bool {
        let elements = settableElements(id, kAudioDevicePropertyMute)
        return !elements.isEmpty && elements.allSatisfy {
            (read(id, kAudioDevicePropertyMute, element: $0) as UInt32?) == 1
        }
    }

    @discardableResult
    public func setMuted(_ id: DeviceID, _ muted: Bool) -> Bool {
        let elements = settableElements(id, kAudioDevicePropertyMute)
        let value: UInt32 = muted ? 1 : 0
        return !elements.isEmpty && elements.allSatisfy { write(id, kAudioDevicePropertyMute, element: $0, value: value) }
    }

    public func canSetVolume(_ id: DeviceID) -> Bool {
        !settableElements(id, kAudioDevicePropertyVolumeScalar).isEmpty
    }

    public func volume(_ id: DeviceID) -> Float {
        settableElements(id, kAudioDevicePropertyVolumeScalar)
            .compactMap { read(id, kAudioDevicePropertyVolumeScalar, element: $0) as Float32? }
            .max() ?? 0
    }

    @discardableResult
    public func setVolume(_ id: DeviceID, _ volume: Float) -> Bool {
        let elements = settableElements(id, kAudioDevicePropertyVolumeScalar)
        return !elements.isEmpty && elements.allSatisfy {
            write(id, kAudioDevicePropertyVolumeScalar, element: $0, value: Float32(volume))
        }
    }

    private func handleChange() {
        watchInputDevices()
        onChange?()
    }

    /// Adds mute/volume listeners to input devices we haven't seen yet. Listeners on removed
    /// devices disappear with the device, so we only need to forget their IDs.
    private func watchInputDevices() {
        let current = Set(inputDeviceIDs())
        for id in current.subtracting(watchedDevices) {
            for selector in [kAudioDevicePropertyMute, kAudioDevicePropertyVolumeScalar] {
                var address = Self.address(selector, element: kAudioObjectPropertyElementWildcard)
                AudioObjectAddPropertyListenerBlock(id, &address, .main, listener)
            }
        }
        watchedDevices = current
    }

    private static func address(
        _ selector: AudioObjectPropertySelector,
        scope: AudioObjectPropertyScope = kAudioDevicePropertyScopeInput,
        element: AudioObjectPropertyElement = kAudioObjectPropertyElementMain
    ) -> AudioObjectPropertyAddress {
        AudioObjectPropertyAddress(mSelector: selector, mScope: scope, mElement: element)
    }

    private func allDeviceIDs() -> [DeviceID] {
        let system = AudioObjectID(kAudioObjectSystemObject)
        var address = Self.address(kAudioHardwarePropertyDevices, scope: kAudioObjectPropertyScopeGlobal)
        var size: UInt32 = 0
        guard AudioObjectGetPropertyDataSize(system, &address, 0, nil, &size) == noErr else { return [] }
        var ids = [AudioObjectID](repeating: 0, count: Int(size) / MemoryLayout<AudioObjectID>.size)
        guard AudioObjectGetPropertyData(system, &address, 0, nil, &size, &ids) == noErr else { return [] }
        return ids
    }

    private func inputChannelCount(_ id: DeviceID) -> Int {
        var address = Self.address(kAudioDevicePropertyStreamConfiguration)
        var size: UInt32 = 0
        guard AudioObjectGetPropertyDataSize(id, &address, 0, nil, &size) == noErr, size > 0 else { return 0 }
        let raw = UnsafeMutableRawPointer.allocate(byteCount: Int(size), alignment: MemoryLayout<AudioBufferList>.alignment)
        defer { raw.deallocate() }
        guard AudioObjectGetPropertyData(id, &address, 0, nil, &size, raw) == noErr else { return 0 }
        let buffers = UnsafeMutableAudioBufferListPointer(raw.assumingMemoryBound(to: AudioBufferList.self))
        return buffers.reduce(0) { $0 + Int($1.mNumberChannels) }
    }

    private func settableElements(_ id: DeviceID, _ selector: AudioObjectPropertySelector) -> [AudioObjectPropertyElement] {
        if isSettable(id, selector, kAudioObjectPropertyElementMain) { return [kAudioObjectPropertyElementMain] }
        let channels = inputChannelCount(id)
        guard channels > 0 else { return [] }
        return (1...channels).map { AudioObjectPropertyElement($0) }.filter { isSettable(id, selector, $0) }
    }

    private func isSettable(_ id: DeviceID, _ selector: AudioObjectPropertySelector, _ element: AudioObjectPropertyElement) -> Bool {
        var address = Self.address(selector, element: element)
        guard AudioObjectHasProperty(id, &address) else { return false }
        var settable: DarwinBoolean = false
        return AudioObjectIsPropertySettable(id, &address, &settable) == noErr && settable.boolValue
    }

    private func read<T: Numeric>(_ id: DeviceID, _ selector: AudioObjectPropertySelector, element: AudioObjectPropertyElement) -> T? {
        var address = Self.address(selector, element: element)
        var value: T = 0
        var size = UInt32(MemoryLayout<T>.size)
        guard AudioObjectGetPropertyData(id, &address, 0, nil, &size, &value) == noErr else { return nil }
        return value
    }

    private func write<T: Numeric>(_ id: DeviceID, _ selector: AudioObjectPropertySelector, element: AudioObjectPropertyElement, value: T) -> Bool {
        var address = Self.address(selector, element: element)
        var value = value
        let status = AudioObjectSetPropertyData(id, &address, 0, nil, UInt32(MemoryLayout<T>.size), &value)
        if status != noErr { log.error("Setting property \(selector) on device \(id) failed: \(status)") }
        return status == noErr
    }

    private func string(_ id: DeviceID, _ selector: AudioObjectPropertySelector) -> String? {
        var address = Self.address(selector, scope: kAudioObjectPropertyScopeGlobal)
        var value: Unmanaged<CFString>?
        var size = UInt32(MemoryLayout<Unmanaged<CFString>?>.size)
        guard AudioObjectGetPropertyData(id, &address, 0, nil, &size, &value) == noErr, let value else { return nil }
        return value.takeRetainedValue() as String
    }
}
```

```swift
import Carbon.HIToolbox

/// A global keyboard shortcut: a virtual key code plus Carbon modifier mask.
public struct Shortcut: Codable, Equatable, Sendable {
    public var keyCode: UInt32
    public var carbonModifiers: UInt32

    public init(keyCode: UInt32, carbonModifiers: UInt32) {
        self.keyCode = keyCode
        self.carbonModifiers = carbonModifiers
    }

    /// - Parameter cocoaModifiers: `NSEvent.ModifierFlags.rawValue`.
    public init(keyCode: UInt32, cocoaModifiers: UInt) {
        var modifiers: UInt32 = 0
        if cocoaModifiers & (1 << 18) != 0 { modifiers |= UInt32(controlKey) }
        if cocoaModifiers & (1 << 19) != 0 { modifiers |= UInt32(optionKey) }
        if cocoaModifiers & (1 << 17) != 0 { modifiers |= UInt32(shiftKey) }
        if cocoaModifiers & (1 << 20) != 0 { modifiers |= UInt32(cmdKey) }
        self.init(keyCode: keyCode, carbonModifiers: modifiers)
    }

    public static let defaultToggle = Shortcut(keyCode: UInt32(kVK_ANSI_M), carbonModifiers: UInt32(controlKey | optionKey))
    public static let defaultPushToTalk = Shortcut(keyCode: UInt32(kVK_Space), carbonModifiers: UInt32(controlKey | optionKey))

    /// Plain keys without modifiers would swallow normal typing; function keys are fine alone.
    public var isValid: Bool {
        carbonModifiers != 0 || Self.functionKeyNames[Int(keyCode)] != nil
    }

    public var modifierSymbols: String {
        var symbols = ""
        if carbonModifiers & UInt32(controlKey) != 0 { symbols += "⌃" }
        if carbonModifiers & UInt32(optionKey) != 0 { symbols += "⌥" }
        if carbonModifiers & UInt32(shiftKey) != 0 { symbols += "⇧" }
        if carbonModifiers & UInt32(cmdKey) != 0 { symbols += "⌘" }
        return symbols
    }

    public var displayString: String { modifierSymbols + Self.keyName(for: keyCode) }

    private static let specialKeyNames: [Int: String] = [
        kVK_Space: "Space", kVK_Return: "↩", kVK_Tab: "⇥", kVK_Delete: "⌫", kVK_ForwardDelete: "⌦",
        kVK_Escape: "⎋", kVK_LeftArrow: "←", kVK_RightArrow: "→", kVK_UpArrow: "↑", kVK_DownArrow: "↓",
        kVK_Home: "↖", kVK_End: "↘", kVK_PageUp: "⇞", kVK_PageDown: "⇟",
    ]

    private static let functionKeyNames: [Int: String] = [
        kVK_F1: "F1", kVK_F2: "F2", kVK_F3: "F3", kVK_F4: "F4", kVK_F5: "F5", kVK_F6: "F6",
        kVK_F7: "F7", kVK_F8: "F8", kVK_F9: "F9", kVK_F10: "F10", kVK_F11: "F11", kVK_F12: "F12",
        kVK_F13: "F13", kVK_F14: "F14", kVK_F15: "F15", kVK_F16: "F16", kVK_F17: "F17", kVK_F18: "F18",
        kVK_F19: "F19",
    ]

    private static func keyName(for keyCode: UInt32) -> String {
        let code = Int(keyCode)
        if let name = specialKeyNames[code] ?? functionKeyNames[code] { return name }
        return layoutCharacter(for: keyCode) ?? "#\(keyCode)"
    }

    /// Character the key produces on the current keyboard layout (e.g. "M" on QWERTY).
    private static func layoutCharacter(for keyCode: UInt32) -> String? {
        guard let source = TISCopyCurrentKeyboardLayoutInputSource()?.takeRetainedValue(),
              let layoutPointer = TISGetInputSourceProperty(source, kTISPropertyUnicodeKeyLayoutData)
        else { return nil }
        let layoutData = Unmanaged<CFData>.fromOpaque(layoutPointer).takeUnretainedValue() as Data
        var deadKeyState: UInt32 = 0
        var chars = [UniChar](repeating: 0, count: 4)
        var length = 0
        let status = layoutData.withUnsafeBytes { raw -> OSStatus in
            guard let layout = raw.bindMemory(to: UCKeyboardLayout.self).baseAddress else { return -1 }
            return UCKeyTranslate(layout, UInt16(keyCode), UInt16(kUCKeyActionDisplay), 0, UInt32(LMGetKbdType()),
                                  OptionBits(kUCKeyTranslateNoDeadKeysBit), &deadKeyState, chars.count, &length, &chars)
        }
        guard status == noErr, length > 0 else { return nil }
        return String(utf16CodeUnits: chars, count: length).uppercased()
    }
}
```

```swift
import Foundation

/// Commands a bridge client (the Stream Deck plugin) can send.
public enum BridgeCommand: Equatable, Sendable {
    case toggle, mute, unmute, get
    case ptt(down: Bool)

    private struct Wire: Decodable {
        let type: String
        let down: Bool?
    }

    public static func parse(_ data: Data) -> BridgeCommand? {
        guard let wire = try? JSONDecoder().decode(Wire.self, from: data) else { return nil }
        switch wire.type {
        case "toggle": return .toggle
        case "mute": return .mute
        case "unmute": return .unmute
        case "get": return .get
        case "ptt": return wire.down.map { .ptt(down: $0) }
        default: return nil
        }
    }
}

/// Messages the app sends to bridge clients.
public enum BridgeMessage {
    private struct State: Encodable {
        let type = "state"
        let muted: Bool
        let pttActive: Bool
    }

    public static func state(_ snapshot: MuteState.Snapshot) -> Data {
        let encoder = JSONEncoder()
        encoder.outputFormatting = .sortedKeys
        return (try? encoder.encode(State(muted: snapshot.muted, pttActive: snapshot.pttActive))) ?? Data()
    }
}
```

```swift
/// Applies bridge commands to MuteState. Each client is its own push-to-talk source,
/// so a client that disconnects mid-hold releases only its own hold.
@MainActor
public final class CommandRouter {
    private let state: MuteState

    public init(state: MuteState) {
        self.state = state
    }

    public func handle(_ command: BridgeCommand, from client: AnyHashable) {
        switch command {
        case .toggle: state.toggle()
        case .mute: state.setMuted(true)
        case .unmute: state.setMuted(false)
        case .ptt(let down): down ? state.pttBegin(source: client) : state.pttEnd(source: client)
        case .get: break // BridgeServer answers with the current state itself.
        }
    }

    public func clientDisconnected(_ client: AnyHashable) {
        state.pttEnd(source: client)
    }
}
```

```swift
/// Loopback-only WebSocket server the Stream Deck plugin connects to.
/// All state is confined to `queue`; callbacks are invoked on it too.
public final class BridgeServer {
    public static let defaultPort: UInt16 = 52836

    public var onCommand: ((BridgeCommand, ObjectIdentifier) -> Void)?
    public var onClientDisconnected: ((ObjectIdentifier) -> Void)?
    public var onStatusChange: ((Bool) -> Void)?

    private let log = Logger(subsystem: "xyz.kayvan.micmute", category: "bridge")
    private let requestedPort: UInt16
    private let queue: DispatchQueue
    private var listener: NWListener?
    private var connections: [ObjectIdentifier: NWConnection] = [:]
    private var currentState: Data

    public init(port: UInt16 = defaultPort, initialState: Data, queue: DispatchQueue = .main) {
        requestedPort = port
        currentState = initialState
        self.queue = queue
    }

    public var port: UInt16? { listener?.port?.rawValue }

    public func start() throws {
        let websocket = NWProtocolWebSocket.Options()
        websocket.autoReplyPing = true
        let parameters = NWParameters.tcp
        parameters.defaultProtocolStack.applicationProtocols.insert(websocket, at: 0)
        parameters.requiredInterfaceType = .loopback
        parameters.allowLocalEndpointReuse = true

        let port = requestedPort == 0 ? NWEndpoint.Port.any : NWEndpoint.Port(rawValue: requestedPort)!
        let listener = try NWListener(using: parameters, on: port)
        listener.newConnectionHandler = { [weak self] connection in self?.accept(connection) }
        listener.stateUpdateHandler = { [weak self] state in
            guard let self else { return }
            switch state {
            case .ready:
                onStatusChange?(true)
            case .failed(let error):
                log.error("Bridge listener failed: \(error.localizedDescription)")
                onStatusChange?(false)
                listener.cancel()
            default:
                break
            }
        }
        self.listener = listener
        listener.start(queue: queue)
    }

    public func stop() {
        queue.async { [self] in
            listener?.cancel()
            listener = nil
            connections.values.forEach { $0.cancel() }
            connections.removeAll()
        }
    }

    public func broadcast(_ state: Data) {
        queue.async { [self] in
            currentState = state
            connections.values.forEach { send(state, to: $0) }
        }
    }

    // MARK: - Connections

    private func accept(_ connection: NWConnection) {
        let id = ObjectIdentifier(connection)
        connections[id] = connection
        connection.stateUpdateHandler = { [weak self, weak connection] state in
            guard let self, let connection else { return }
            switch state {
            case .ready: send(currentState, to: connection)
            case .failed, .cancelled: drop(id)
            default: break
            }
        }
        connection.start(queue: queue)
        receive(on: connection)
    }

    private func receive(on connection: NWConnection) {
        let id = ObjectIdentifier(connection)
        connection.receiveMessage { [weak self] data, context, _, error in
            guard let self else { return }
            let metadata = context?.protocolMetadata(definition: NWProtocolWebSocket.definition) as? NWProtocolWebSocket.Metadata
            if error != nil || metadata?.opcode == .close {
                drop(id)
                return
            }
            if let data, let command = BridgeCommand.parse(data) {
                if command == .get { send(currentState, to: connection) }
                onCommand?(command, id)
            }
            receive(on: connection)
        }
    }

    private func drop(_ id: ObjectIdentifier) {
        guard let connection = connections.removeValue(forKey: id) else { return }
        connection.cancel()
        onClientDisconnected?(id)
    }

    private func send(_ data: Data, to connection: NWConnection) {
        let metadata = NWProtocolWebSocket.Metadata(opcode: .text)
        let context = NWConnection.ContentContext(identifier: "state", metadata: [metadata])
        connection.send(content: data, contentContext: context, isComplete: true, completion: .idempotent)
    }
}
```

```swift
        let muted = state.snapshot.muted
        let label = muted ? "Microphones muted" : "Microphones live"
        var image = NSImage(systemSymbolName: muted ? "mic.slash.fill" : "mic.fill", accessibilityDescription: label)

        let flags = event?.modifierFlags ?? []

            showMenu()
        } else {
            state.toggle()
        }
    }

    private func showMenu() {
        item.menu = buildMenu()
        item.button?.performClick(nil)
        item.menu = nil
    }

    private func buildMenu() -> NSMenu {

        let muted = state.snapshot.muted
        menu.addItem(disabledItem(muted ? "Microphones: Muted" : "Microphones: Live"))

        menu.addItem(disabledItem("Input Devices"))

            menu.addItem(disabledItem("⚠︎ Stream Deck bridge unavailable"))
```

```swift
import Carbon.HIToolbox
import os

/// System-wide hotkeys via Carbon `RegisterEventHotKey`. Delivers both key-down and key-up
/// (needed for push-to-talk) and needs no Accessibility permission.
@MainActor
final class HotkeyManager {
    private struct Handlers {
        let onDown: () -> Void
        let onUp: () -> Void
    }

    private static let signature: OSType = 0x4D4D_5554 // 'MMUT'
    private let log = Logger(subsystem: "xyz.kayvan.micmute", category: "hotkeys")
    private var refs: [UInt32: EventHotKeyRef] = [:]
    private var handlers: [UInt32: Handlers] = [:]
    private var pressed: Set<UInt32> = []
    private var eventHandler: EventHandlerRef?

    init() {
        var eventTypes = [
            EventTypeSpec(eventClass: OSType(kEventClassKeyboard), eventKind: UInt32(kEventHotKeyPressed)),
            EventTypeSpec(eventClass: OSType(kEventClassKeyboard), eventKind: UInt32(kEventHotKeyReleased)),
        ]
        let context = Unmanaged.passUnretained(self).toOpaque()
        let status = InstallEventHandler(GetApplicationEventTarget(), { _, event, context in
            guard let event, let context else { return OSStatus(eventNotHandledErr) }
            var hotKeyID = EventHotKeyID()
            GetEventParameter(event, EventParamName(kEventParamDirectObject), EventParamType(typeEventHotKeyID),
                              nil, MemoryLayout<EventHotKeyID>.size, nil, &hotKeyID)
            let isDown = GetEventKind(event) == UInt32(kEventHotKeyPressed)
            let manager = Unmanaged<HotkeyManager>.fromOpaque(context).takeUnretainedValue()
            MainActor.assumeIsolated { manager.handle(id: hotKeyID.id, isDown: isDown) }
            return noErr
        }, eventTypes.count, &eventTypes, context, &eventHandler)
        if status != noErr { log.error("InstallEventHandler failed: \(status)") }
    }

    @discardableResult
    func register(id: UInt32, shortcut: Shortcut, onDown: @escaping () -> Void, onUp: @escaping () -> Void) -> Bool {
        unregister(id: id)
        var ref: EventHotKeyRef?
        let hotKeyID = EventHotKeyID(signature: Self.signature, id: id)
        let status = RegisterEventHotKey(shortcut.keyCode, shortcut.carbonModifiers, hotKeyID,
                                         GetApplicationEventTarget(), 0, &ref)
        guard status == noErr, let ref else {
            log.error("RegisterEventHotKey \(shortcut.displayString) failed: \(status)")
            return false
        }
        refs[id] = ref
        handlers[id] = Handlers(onDown: onDown, onUp: onUp)
        return true
    }

    func unregister(id: UInt32) {
        if let ref = refs.removeValue(forKey: id) { UnregisterEventHotKey(ref) }
        let removed = handlers.removeValue(forKey: id)
        // A hold in progress must still be released, or push-to-talk would stick.
        if pressed.remove(id) != nil { removed?.onUp() }
    }

    func unregisterAll() {
        Array(refs.keys).forEach { unregister(id: $0) }
    }

    private func handle(id: UInt32, isDown: Bool) {
        guard let handler = handlers[id] else { return }
        if isDown {
            // Ignore key auto-repeat so a held toggle key doesn't flap.
            guard pressed.insert(id).inserted else { return }
            handler.onDown()
        } else {
            guard pressed.remove(id) != nil else { return }
            handler.onUp()
        }
    }
}
```

```swift
        .frame(minWidth: 150)
        .onDisappear { stop() }

    private func start() {
        recording = true
        onRecordingChanged(true)
        monitor = NSEvent.addLocalMonitorForEvents(matching: .keyDown) { event in
            if event.keyCode == UInt16(kVK_Escape) {
                stop()
                return nil
            }
            let flags = event.modifierFlags.intersection(.deviceIndependentFlagsMask)
            let candidate = Shortcut(keyCode: UInt32(event.keyCode), cocoaModifiers: flags.rawValue)
            guard candidate.isValid, candidate != conflict else {
                NSSound.beep()
                return nil
            }
            onChange(candidate)
            stop()
            return nil
        }
    }

    private func stop() {
        guard recording else { return }
        if let monitor { NSEvent.removeMonitor(monitor) }
        monitor = nil
        recording = false
```

```ts
export type MicState = { muted: boolean; pttActive: boolean };

export type Command =
  | { type: "toggle" }
  | { type: "mute" }
  | { type: "unmute" }
  | { type: "get" }
  | { type: "ptt"; down: boolean };

/** The subset of the WebSocket API BridgeClient uses (lets tests inject a fake). */
export interface SocketLike {
  readonly readyState: number;
  onopen: (() => void) | null;
  onclose: (() => void) | null;
  onerror: (() => void) | null;
  onmessage: ((event: { data: unknown }) => void) | null;
  send(data: string): void;
  close(): void;
}

export type SocketFactory = (url: string) => SocketLike;

const OPEN = 1;

export function parseState(raw: string): MicState | null {
  try {
    const message = JSON.parse(raw);
    if (message?.type === "state" && typeof message.muted === "boolean") {
      return { muted: message.muted, pttActive: message.pttActive === true };
    }
  } catch {
    // Not JSON; ignore.
  }
  return null;
}

/** Keeps a WebSocket connection to the MicMute app, reconnecting while it isn't running. */
export class BridgeClient {
  /** Latest state from the app, or null while the app is unreachable. */
  state: MicState | null = null;

  private readonly url: string;
  private readonly retryMs: number;
  private readonly createSocket: SocketFactory;
  private readonly listeners = new Set<(state: MicState | null) => void>();
  private socket?: SocketLike;
  private retryTimer?: ReturnType<typeof setTimeout>;
  private stopped = false;

  constructor(options: { url?: string; retryMs?: number; createSocket?: SocketFactory } = {}) {
    this.url = options.url ?? "ws://127.0.0.1:52836";
    this.retryMs = options.retryMs ?? 2000;
    this.createSocket = options.createSocket ?? ((url) => new WebSocket(url) as unknown as SocketLike);
  }

  start(): void {
    this.stopped = false;
    this.connect();
  }

  stop(): void {
    this.stopped = true;
    clearTimeout(this.retryTimer);
    this.retryTimer = undefined;
    this.socket?.close();
  }

  onChange(listener: (state: MicState | null) => void): () => void {
    this.listeners.add(listener);
    return () => this.listeners.delete(listener);
  }

  send(command: Command): boolean {
    if (this.socket?.readyState !== OPEN) return false;
    this.socket.send(JSON.stringify(command));
    return true;
  }

  private connect(): void {
    const socket = this.createSocket(this.url);
    this.socket = socket;
    socket.onmessage = (event) => {
      const state = parseState(String(event.data));
      if (state) this.setState(state);
    };
    socket.onclose = () => {
      if (this.socket === socket) this.socket = undefined;
      this.setState(null);
      this.scheduleReconnect();
    };
    socket.onerror = () => {
      // A close event always follows; reconnect happens there.
    };
  }

  private scheduleReconnect(): void {
    if (this.stopped || this.retryTimer) return;
    this.retryTimer = setTimeout(() => {
      this.retryTimer = undefined;
      this.connect();
    }, this.retryMs);
  }

  private setState(state: MicState | null): void {
    if (state === null && this.state === null) return;
    this.state = state;
    this.listeners.forEach((listener) => listener(state));
  }
}
```

```swift
import AppKit

func render(pixels: Int, to path: String) throws {
    let size = CGFloat(pixels)
    guard let rep = NSBitmapImageRep(bitmapDataPlanes: nil, pixelsWide: pixels, pixelsHigh: pixels,
                                     bitsPerSample: 8, samplesPerPixel: 4, hasAlpha: true, isPlanar: false,
                                     colorSpaceName: .deviceRGB, bytesPerRow: 0, bitsPerPixel: 0) else { exit(1) }
    NSGraphicsContext.saveGraphicsState()
    NSGraphicsContext.current = NSGraphicsContext(bitmapImageRep: rep)
    NSColor(calibratedRed: 0.86, green: 0.15, blue: 0.15, alpha: 1).setFill()
    NSBezierPath(roundedRect: NSRect(x: 0, y: 0, width: size, height: size), xRadius: size * 0.22, yRadius: size * 0.22).fill()
    let config = NSImage.SymbolConfiguration(pointSize: size * 0.5, weight: .semibold)
        .applying(NSImage.SymbolConfiguration(paletteColors: [.white]))
    if let mic = NSImage(systemSymbolName: "mic.fill", accessibilityDescription: nil)?.withSymbolConfiguration(config) {
        let s = mic.size
        mic.draw(in: NSRect(x: (size - s.width) / 2, y: (size - s.height) / 2, width: s.width, height: s.height))
    }
    NSGraphicsContext.restoreGraphicsState()
    guard let png = rep.representation(using: .png, properties: [:]) else { exit(1) }
    try png.write(to: URL(fileURLWithPath: path))
}
```

````markdown
# MicMute

A macOS menu bar app that mutes **every** microphone at once — built-in, USB, Bluetooth — with global hotkeys and a Stream Deck plugin.

## Features

- Menu bar icon: left-click toggles, right-click shows devices, Settings and Quit.
- Mutes all input devices; mics plugged in while muted are muted immediately. Devices without a hardware mute control are muted by dropping their input volume to 0 (restored on unmute).
- Global shortcuts (configurable): **⌃⌥M** toggle, **⌃⌥Space** hold-to-talk.
- Optional mute/unmute sounds, launch at login.
- Stream Deck: *Toggle Mute* and *Push to Talk* keys that show live state.
- Quitting unmutes everything.

## Requirements

- macOS 13+, Swift toolchain (Command Line Tools are enough; Xcode not required).
- Stream Deck 7.1+ and Node.js 20+ (for building the plugin).

## Build & install

```bash
make test       # unit tests
make install    # builds build/MicMute.app, copies to /Applications, launches
```

`MicMute --status` (`/Applications/MicMute.app/Contents/MacOS/MicMute --status`) lists each input device and how it's muted.

## Stream Deck plugin

```bash
cd streamdeck && npm install && cd ..
make plugin-test
make plugin-link   # build, validate, link into Stream Deck and restart the plugin
```

`cd streamdeck && npm run pack` produces a distributable `.streamDeckPlugin` in `streamdeck/dist/`.

The plugin talks to the app over a WebSocket on `127.0.0.1:52836` (loopback only). For debugging: `node scripts/bridge.mjs [get|toggle|mute|unmute|ptt down|ptt up]`.

## Layout

- `Sources/MicMuteCore` — CoreAudio wrapper, mute logic, preferences, WebSocket bridge (unit tested).
- `Sources/MicMute` — AppKit/SwiftUI app shell.
- `streamdeck/` — TypeScript Stream Deck plugin.
````
